1. Core Syntax & Modules

PIXEL-BASIC is a strictly-typed, procedural language. It supports multi-file architecture via imports and isolated lexical scopes.

  • Variables: Use explicit typing (: type) or type inference (:=).
  • Subroutines: Define rigid signatures mapping input parameters to specific return types.
REM physics.basic
EXPORT CONST gravity: f32 = 9.8
EXPORT LET terminal_velocity := 50.0

EXPORT SUB apply_gravity current_v: f32 : f32 THEN
  RETURN current_v + gravity
END SUB

REM main.basic
IMPORT gravity, apply_gravity FROM "physics.basic"

2. Graphics & Canvas API

The engine utilizes an immediate-mode 2D rendering pipeline bridged directly to HTML5 Canvas.

  • Setup: Initialize screen dimensions using SCREEN. Clear the buffer using CLEAR_SCREEN.
  • Styling: Control draw states with FILL_COLOR, STROKE_COLOR, and STROKE_WEIGHT.
REM Initialize an 800x600 canvas
SCREEN 800, 600

LET is_running := TRUE

WHILE is_running == TRUE THEN
  CLEAR_SCREEN
  
  REM Draw a solid red background
  FILL_COLOR 255, 0, 0
  DRAW_RECT 0, 0, SCR_W, SCR_H
  
  REM Draw a green outlined circle in the center
  NO_FILL
  STROKE_COLOR 0, 255, 0
  STROKE_WEIGHT 4
  DRAW_CIRCLE 400, 300, 50
END WHILE

3. Mathematics & System State

PIXEL-BASIC includes standard mathematical operations and real-time polling for hardware inputs.

  • Constants: PI, TWO_PI, HALF_PI, SCR_W, SCR_H.
  • Math API: SQRT(), POW(), SIN(), COS(), TAN(), ATAN2(), CLAMP(), LERP(), RND().
REM Track mouse position smoothly using Linear Interpolation
LET ship_x := 0.0
LET ship_y := 0.0

WHILE TRUE THEN
  ship_x = LERP(ship_x, MOUSE_X, 0.1)
  ship_y = LERP(ship_y, MOUSE_Y, 0.1)

  REM Generate random particles while Spacebar is held
  IF IS_KEY_DOWN(" ") THEN
    LET px = ship_x + RND(-10, 10)
    LET py = ship_y + RND(-10, 10)
    DRAW_RECT px, py, 2, 2
  END IF
END WHILE

4. Asset Integration

Assets are managed locally via IndexedDB. To use an asset, first upload it using the + icon in the editor's ASSETS sidebar.

  • Images: Supports .png and .jpg. Call DRAW_IMAGE using the exact filename string.
  • Typography: Supports .ttf, .otf, and .woff. Setup the font using SET_FONT minus the extension.
REM Ensure 'hero.png' and 'retro.ttf' are uploaded in the sidebar

WHILE TRUE THEN
  CLEAR_SCREEN

  REM Draw the uploaded image (filename, x, y, width, height)
  DRAW_IMAGE "hero.png", 100, 100, 64, 64

  REM Apply the uploaded font and draw text
  SET_FONT 24, "retro"
  TEXT_ALIGN "center"
  FILL_COLOR 255, 255, 255
  
  DRAW_TEXT SCR_W / 2, 50, "READY PLAYER ONE"
END WHILE
